import json, sys
def runtime(path):
    code = bytes.fromhex(json.load(open(path))["deployedBytecode"]["object"].removeprefix("0x"))
    cut = int.from_bytes(code[-2:], "big") + 2  # CBOR metadata length plus its 2-byte length field
    return code[:-cut], code[-cut:]
a, ma = runtime(sys.argv[1])
b, mb = runtime(sys.argv[2])
print("poc build    runtime bytes without metadata", len(a))
print("contracts/   runtime bytes without metadata", len(b))

print("metadata identical:", ma == mb)
