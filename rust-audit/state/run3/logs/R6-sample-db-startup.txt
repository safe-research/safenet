## mockrpc.py
"""Minimal loopback JSON-RPC stub for R6 startup checks.

Answers eth_chainId with 0x64 (Gnosis); every other method gets a JSON-RPC
error. Binds 127.0.0.1 on an OS-assigned port and writes the port to argv[1].
Exits after argv[2] seconds.
"""

import json
import sys
import threading
from http.server import BaseHTTPRequestHandler, HTTPServer


class Handler(BaseHTTPRequestHandler):
    def do_POST(self):
        body = self.rfile.read(int(self.headers.get("content-length", "0")))
        req = json.loads(body)

        def answer(r):
            if r.get("method") == "eth_chainId":
                return {"jsonrpc": "2.0", "id": r.get("id"), "result": "0x64"}
            return {
                "jsonrpc": "2.0",
                "id": r.get("id"),
                "error": {"code": -32601, "message": "stub: method not served"},
            }

        out = [answer(r) for r in req] if isinstance(req, list) else answer(req)
        data = json.dumps(out).encode()
        self.send_response(200)
        self.send_header("content-type", "application/json")
        self.send_header("content-length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)
        sys.stderr.write("stub served %s\n" % (req.get("method") if isinstance(req, dict) else "batch"))

    def log_message(self, *args):
        pass


server = HTTPServer(("127.0.0.1", 0), Handler)
with open(sys.argv[1], "w") as f:
    f.write(str(server.server_address[1]))
threading.Timer(float(sys.argv[2]), server.shutdown).start()
server.serve_forever()
