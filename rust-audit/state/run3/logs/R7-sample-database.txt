/// R7 (run 3): the sample `database` URL form on a fresh node. Appended to a scratch copy of
/// `config.rs` only.
#[cfg(test)]
mod r7_poc_config {
    use super::*;

    #[tokio::test]
    async fn sample_database_url_cannot_create_a_fresh_database() {
        let path = Path::new(env!("CARGO_MANIFEST_DIR")).join("sentinel.sample.toml");
        let sample = std::fs::read_to_string(path).unwrap();
        let config = toml::from_str::<Config>(&sample).unwrap();
        println!("sample database filename: {}", config.database.get_filename().display());

        // Same URL form as the sample (`sqlite:<absolute path>`), in a fresh, empty, writable
        // directory, as on a new volume.
        let dir = std::env::temp_dir().join(format!("r7-sqlite-{}", std::process::id()));
        std::fs::create_dir_all(&dir).unwrap();
        let db = dir.join("storage.db");
        let sample_form = sample.replace(
            "sqlite:/var/lib/safenet/sentinel/data/storage.db",
            &format!("sqlite:{}", db.display()),
        );
        assert_ne!(sample_form, sample, "sample database line not found");
        let config = toml::from_str::<Config>(&sample_form).unwrap();
        let result = safenet_core::utils::connect_sqlite(config.database).await;
        println!("connect with the sample URL form: {:?}", result.as_ref().map(|_| "ok").map_err(|err| err.to_string()));
        assert!(result.is_err());
        assert!(!db.exists());

        let with_mode = sample.replace(
            "sqlite:/var/lib/safenet/sentinel/data/storage.db",
            &format!("sqlite:{}?mode=rwc", db.display()),
        );
        let config = toml::from_str::<Config>(&with_mode).unwrap();
        let result = safenet_core::utils::connect_sqlite(config.database).await;
        println!("connect with ?mode=rwc: {:?}", result.as_ref().map(|_| "ok").map_err(|err| err.to_string()));
        assert!(result.is_ok());
        assert!(db.exists());
    }
}
