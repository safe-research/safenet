// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.30;

import {Test, Vm, console2} from "forge-std/Test.sol";
import {UsdcCode} from "../src/UsdcCode.sol";

interface IERC20 {
    function approve(address spender, uint256 amount) external returns (bool);
    function allowance(address owner, address spender) external view returns (uint256);
    function symbol() external view returns (string memory);
}

/// Measures the gas of `approve` on the real Gnosis USDC.e bytecode (FiatTokenProxy and
/// FiatTokenV2_2 runtime code and proxy storage from Foundry's RPC cache at Gnosis block
/// 48635154) against the sentinel's hard-coded `ApproveToken` gas limit.
contract UsdcApproveGasTest is Test {
    // Aegis SentinelOracle (spender of the bond) and an arbitrary sentinel EOA.
    address constant ORACLE = 0x4F61B8832978e83b80D69551AEf07557DBE41d03;
    address constant SENTINEL = 0x00000000000000000000000000000000005E471e;
    // Aegis bond: fee 400000 (0.40 USDC.e) * bond multiplier 2000.
    uint256 constant BOND = 800_000_000;
    // `SentinelActionKind::ApproveToken` gas, crates/sentinel/src/service.rs:876.
    uint256 constant APPROVE_GAS = 55_000;

    function setUp() public {
        vm.etch(UsdcCode.PROXY, UsdcCode.proxyCode());
        vm.etch(UsdcCode.IMPL, UsdcCode.implCode());
        (bytes32[] memory keys, bytes32[] memory vals) = UsdcCode.slots();
        for (uint256 i; i < keys.length; i++) {
            vm.store(UsdcCode.PROXY, keys[i], vals[i]);
        }
    }

    function _data() internal pure returns (bytes memory) {
        return abi.encodeCall(IERC20.approve, (ORACLE, BOND));
    }

    /// EIP-2028 intrinsic gas of a plain call transaction carrying `data`.
    function _intrinsic(bytes memory data) internal pure returns (uint256 gas) {
        gas = 21_000;
        for (uint256 i; i < data.length; i++) {
            gas += data[i] == 0 ? 4 : 16;
        }
    }

    function test_cachedCodeMatchesOnchainCodeHashes() public view {
        assertEq(keccak256(UsdcCode.PROXY.code), UsdcCode.PROXY_CODE_HASH);
        assertEq(keccak256(UsdcCode.IMPL.code), UsdcCode.IMPL_CODE_HASH);
        // The proxy reaches the implementation and reads the cached storage: `paused()` is
        // false and `blacklister()` is the cached slot 2 value.
        (bool ok, bytes memory ret) = UsdcCode.PROXY.staticcall(abi.encodeWithSignature("paused()"));
        assertTrue(ok);
        assertEq(abi.decode(ret, (bool)), false);
        (ok, ret) = UsdcCode.PROXY.staticcall(abi.encodeWithSignature("blacklister()"));
        assertTrue(ok);
        assertEq(abi.decode(ret, (address)), 0x458cD345B4C05e8DF39d0A07220feb4Ec19F5e6f);
    }

    /// Direct mode: the transaction's gas limit is the action's `gas` (55,000), so the token
    /// code runs with 55,000 minus the intrinsic gas. The allowance starts at zero, as it does
    /// after every commit spends an exact approval.
    function test_directModeApproveFromZeroAllowance() public {
        bytes memory data = _data();
        uint256 budget = APPROVE_GAS - _intrinsic(data);
        vm.prank(SENTINEL);
        (bool ok,) = UsdcCode.PROXY.call{gas: budget}(data);
        console2.log("intrinsic gas", _intrinsic(data));
        console2.log("execution budget in Direct mode", budget);
        console2.log("approve from zero allowance succeeded", ok);
        console2.log("allowance after", IERC20(UsdcCode.PROXY).allowance(SENTINEL, ORACLE));
    }

    /// Same budget when an allowance is already set (nonzero to nonzero).
    function test_directModeApproveFromNonzeroAllowance() public {
        // allowed[SENTINEL][ORACLE] = 1 through a first, generously funded approval in setUp-like
        // fashion, then cool the accounts so the measured call starts cold again.
        vm.prank(SENTINEL);
        IERC20(UsdcCode.PROXY).approve(ORACLE, 1);
        vm.cool(UsdcCode.PROXY);
        vm.cool(UsdcCode.IMPL);
        bytes memory data = _data();
        uint256 budget = APPROVE_GAS - _intrinsic(data);
        vm.prank(SENTINEL);
        (bool ok,) = UsdcCode.PROXY.call{gas: budget}(data);
        console2.log("approve from nonzero allowance succeeded", ok);
    }

    /// Exact callee gas of the approval from zero allowance, and the smallest gas limit a
    /// Direct-mode transaction needs.
    function test_measureApproveFromZeroAllowance() public {
        bytes memory data = _data();
        vm.prank(SENTINEL);
        (bool ok,) = UsdcCode.PROXY.call{gas: 200_000}(data);
        assertTrue(ok);
        Vm.Gas memory gas = vm.lastCallGas();
        console2.log("callee gas used (lastCallGas)", gas.gasTotalUsed);
        console2.log("Direct-mode tx gas needed", gas.gasTotalUsed + _intrinsic(data));
    }

    /// Smallest callee gas at which the approval from zero allowance succeeds, found by binary
    /// search with every account cooled before each attempt.
    function test_minimumGasSearch() public {
        bytes memory data = _data();
        uint256 lo = 10_000;
        uint256 hi = 200_000;
        while (lo < hi) {
            uint256 mid = (lo + hi) / 2;
            uint256 id = vm.snapshotState();
            vm.cool(UsdcCode.PROXY);
            vm.cool(UsdcCode.IMPL);
            vm.prank(SENTINEL);
            (bool ok,) = UsdcCode.PROXY.call{gas: mid}(data);
            bool set = IERC20(UsdcCode.PROXY).allowance(SENTINEL, ORACLE) == BOND;
            vm.revertToState(id);
            if (ok && set) {
                hi = mid;
            } else {
                lo = mid + 1;
            }
        }
        console2.log("minimum callee gas (search)", lo);
        console2.log("minimum Direct-mode tx gas limit", lo + _intrinsic(data));
        console2.log("sentinel ApproveToken gas", APPROVE_GAS);
    }

    /// Batched mode: the executor forwards the action's `gas` (55,000) to the token call.
    function test_batchedModeApproveFromZeroAllowance() public {
        bytes memory data = _data();
        vm.prank(SENTINEL);
        (bool ok,) = UsdcCode.PROXY.call{gas: APPROVE_GAS}(data);
        console2.log("approve with the full 55,000 as callee gas succeeded", ok);
    }
}

// SPDX-License-Identifier: UNLICENSED
pragma solidity ^0.8.30;

import {Test, Vm, console2} from "forge-std/Test.sol";
import {UsdcCode} from "../src/UsdcCode.sol";

interface IERC20 {
    function approve(address spender, uint256 amount) external returns (bool);
    function balanceOf(address account) external view returns (uint256);
}

interface ISentinelOracle {
    function postRequest(bytes32 requestId, address sponsor, bytes calldata) external;
    function commit(bytes32 requestId, bytes32 commitHash) external;
    function reveal(bytes32 requestId, bool approve, bytes32 salt, string calldata reason) external;
    function hashCommitment(address sentinel, bytes32 requestId, bool approve, bytes32 salt, string calldata reason)
        external
        pure
        returns (bytes32);
    function finalize(bytes32 requestId) external;
    function claim(bytes32 requestId) external;
    function timeoutArbitration(bytes32 requestId) external;
    function addSentinel(address sentinel) external;
}

/// Same field order and types as `SentinelOracle.ConstructorParams` (contracts/src/SentinelOracle.sol:157-188).
struct ConstructorParams {
    address arbitrator;
    address governance;
    address protocolFundsReceiver;
    address proposer;
    address feeToken;
    uint96 requestFee;
    uint32 initialBondMultiplier;
    uint32 initialSlashingMultiplier;
    uint24 initialDaoFeeShare;
    uint32 commitWindow;
    uint32 revealWindow;
    uint32 governanceDelay;
    uint32 arbitrationTimeout;
    string initialCharterEns;
}

/// Deploys the `SentinelOracle` build artifact (optimizer 1,000,000 runs, via IR, as deployed) with
/// the Aegis parameters and the real USDC.e bytecode, then measures each sentinel action in its
/// own transaction with the Direct-mode budget of the sentinel's fixed gas limit (250,000).
abstract contract Base is Test {
    ISentinelOracle oracle;
    IERC20 constant USDC = IERC20(UsdcCode.PROXY);
    address constant SPONSOR = address(0x5905);
    address constant RECEIVER = address(0x9D02);
    address constant ARBITRATOR = address(0xA4B1);
    address[4] sentinels = [address(0x51), address(0x52), address(0x53), address(0x54)];
    uint256 constant FEE = 400_000;
    uint256 constant BOND = 800_000_000;
    uint256 constant ACTION_GAS = 250_000;
    bytes32 constant ID = keccak256("request");
    uint256 P;

    function setUp() public virtual {
        vm.etch(UsdcCode.PROXY, UsdcCode.proxyCode());
        vm.etch(UsdcCode.IMPL, UsdcCode.implCode());
        (bytes32[] memory keys, bytes32[] memory vals) = UsdcCode.slots();
        for (uint256 i; i < keys.length; i++) {
            vm.store(UsdcCode.PROXY, keys[i], vals[i]);
        }
        ConstructorParams memory params = ConstructorParams({
            arbitrator: ARBITRATOR,
            governance: address(this),
            protocolFundsReceiver: RECEIVER,
            proposer: address(this),
            feeToken: address(USDC),
            requestFee: uint96(FEE),
            initialBondMultiplier: 2000,
            initialSlashingMultiplier: 1,
            initialDaoFeeShare: 0,
            commitWindow: 6,
            revealWindow: 6,
            governanceDelay: 0,
            arbitrationTimeout: 483_840,
            initialCharterEns: "charter.safenet-gov.eth"
        });
        bytes memory code = abi.encodePacked(vm.getCode("artifacts/SentinelOracle.json"), abi.encode(params));
        address deployed;
        assembly {
            deployed := create(0, add(code, 0x20), mload(code))
        }
        require(deployed != address(0), "deploy failed");
        oracle = ISentinelOracle(deployed);
        deal(address(USDC), SPONSOR, FEE);
        vm.prank(SPONSOR);
        USDC.approve(address(oracle), FEE);
        for (uint256 i; i < sentinels.length; i++) {
            oracle.addSentinel(sentinels[i]);
            deal(address(USDC), sentinels[i], BOND);
            vm.prank(sentinels[i]);
            USDC.approve(address(oracle), BOND);
        }
        vm.roll(1_000);
        P = block.number;
        oracle.postRequest(ID, SPONSOR, "");
    }

    function _salt(address s) internal pure returns (bytes32) {
        return keccak256(abi.encode("salt", s));
    }

    function _commit(address s, bool approve, string memory reason) internal {
        bytes32 hash = oracle.hashCommitment(s, ID, approve, _salt(s), reason);
        vm.prank(s);
        oracle.commit(ID, hash);
    }

    function _reveal(address s, bool approve, string memory reason) internal {
        vm.prank(s);
        oracle.reveal(ID, approve, _salt(s), reason);
    }

    function _intrinsic(bytes memory data) internal pure returns (uint256 gas) {
        gas = 21_000;
        for (uint256 i; i < data.length; i++) {
            gas += data[i] == 0 ? 4 : 16;
        }
    }

    /// Calls the oracle as a Direct-mode transaction with gas limit `ACTION_GAS` would run it.
    function _direct(string memory label, address from, bytes memory data) internal returns (bool ok) {
        uint256 budget = ACTION_GAS - _intrinsic(data);
        vm.prank(from);
        (ok,) = address(oracle).call{gas: budget}(data);
        Vm.Gas memory gas = vm.lastCallGas();
        console2.log(label);
        console2.log("  succeeded", ok);
        console2.log("  callee gas used", gas.gasTotalUsed);
        console2.log("  tx gas used incl. intrinsic", gas.gasTotalUsed + _intrinsic(data));
        console2.log("  headroom to 250,000", ACTION_GAS - (gas.gasTotalUsed + _intrinsic(data)));
    }
}
